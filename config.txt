# Configuration for Lock Status Monitor

# Your Discord Bot Token (KEEP THIS SECRET!)
TOKEN=YOUR_DISCORD_BOT_TOKEN_HERE

# The ID of the Discord channel where messages should be sent
CHANNEL_ID=YOUR_DISCORD_CHANNEL_ID_HERE

# --- Display Recovery Settings (Optional) ---

# Path to DesktopOK folder (contains exe and .dok files)
# Download from: https://www.softwareok.com/?seession=DesktopOK
# Example: C:\DesktopOK
DESKTOPOK_PATH=

# Delay in milliseconds before restoring windows after unlock (default: 5000)
# Increase if your monitor takes longer to wake up from sleep
MONITOR_DELAY_MS=5000

# --- Win+L Debounce Settings ---

# Cooldown in milliseconds to prevent Win+L spam from fingerprint reader lock buttons
# Set to 0 to disable. Default is 2000ms (2 seconds)
LOCK_COOLDOWN_MS=2000

# --- Display Standby Enforcement (Optional) ---
# Apps that assert a DISPLAY power request (Moonlight, presentation/slideshow
# windows, media players) keep the monitors lit indefinitely. When enabled, the
# app closes those programs after you lock so the panels can go into standby.
# Master switch. Everything below is ignored unless this is true.
DISPLAY_STANDBY_ENFORCE=false

# Comma-separated process names to close (".exe" optional). Works without admin.
# Example: DISPLAY_KILL_PROCESSES=Moonlight,POWERPNT,vlc
DISPLAY_KILL_PROCESSES=

# Also auto-discover blockers via 'powercfg /requests' and close them.
# REQUIRES the app to run elevated; skipped (with a warning) otherwise.
DISPLAY_KILL_AUTO=false

# Names never closed, on top of the built-in system-process protection list.
DISPLAY_KILL_EXCLUDE=

# Minutes the PC must stay CONTINUOUSLY locked before anything is closed.
# Unlocking during the wait cancels the sweep. Default 30.
DISPLAY_KILL_DELAY_MINUTES=30

# Force-kill a process that ignores the polite close request (default true).
# Set false if you would rather keep unsaved work than guarantee standby.
DISPLAY_KILL_FORCE=true

# After sweeping, tell the monitors to power off immediately. This is the OLED
# safety net for a blocker that could not be closed.
DISPLAY_FORCE_OFF=true
